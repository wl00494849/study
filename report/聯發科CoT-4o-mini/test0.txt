# 步驟 1: 整理資料
df1_selected = df1[['券商', '買進股數', '價格']].copy()
df1_selected.columns = ['券商', '買進股數', '價格']
df1_selected['成交金額'] = df1_selected['買進股數'] * df1_selected['價格']

df2_selected = df2[['券商', '買進股數', '價格']].copy()
df2_selected.columns = ['券商', '買進股數', '價格']
df2_selected['成交金額'] = df2_selected['買進股數'] * df2_selected['價格']

combined = pd.concat([df1_selected, df2_selected])

# 步驟 2: 列出所有券商
grouped = combined.groupby('券商').agg(
    總買進股數=('買進股數', 'sum'),
    總成交金額=('成交金額', 'sum')
).reset_index()

# 步驟 3: 計算均價
grouped['均價'] = grouped['總成交金額'] / grouped['總買進股數']
grouped['均價'] = grouped['均價'].fillna(0).round(1)

# 步驟 4: 整理結果
grouped['股票代號'] = '2454'

# 步驟 5: 整理欄位順序
result = grouped[['券商', '均價', '總買進股數', '總成交金額', '股票代號']]
