```python
df1_cleaned = df1.drop(columns=['Unnamed: 5'])
df2_cleaned = df2.drop(columns=['Unnamed: 5'])

df1_grouped = df1_cleaned.groupby('券商').agg(
    總買進股數=('買進股數', 'sum'),
    總買進成交金額=('價格', lambda x: (x * df1_cleaned.loc[x.index, '買進股數']).sum())
).reset_index()

df2_grouped = df2_cleaned.groupby('券商').agg(
    總買進股數=('買進股數', 'sum'),
    總買進成交金額=('價格', lambda x: (x * df2_cleaned.loc[x.index, '買進股數']).sum())
).reset_index()

merged = pd.merge(df1_grouped, df2_grouped, on='券商', suffixes=('_1', '_2'))
merged['總買進股數'] = merged['總買進股數_1'] + merged['總買進股數_2']
merged['總買進成交金額'] = merged['總買進成交金額_1'] + merged['總買進成交金額_2']
merged['均價'] = merged['總買進成交金額'] / merged['總買進股數']
merged['均價'] = merged['均價'].round(1)

result = merged[['券商', '均價', '總買進股數', '總買進成交金額']]
result['股票代號'] = '2454'
```
