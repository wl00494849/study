4. **排序**：
   - 根據郵遞區號進行升序排序。

5. **最終輸出**：
   - 得到最終合併的 DataFrame，只保留欄位：郵遞區號、地區、熱點名稱、地址、經度、緯度。

```python
df1_selected = df1[['地區', '熱點名稱', '地址', '緯度', '經度']].copy()
df1_selected['郵遞區號'] = None

df2_selected = df2[['地點', '地址', 'Latitude', 'Longitude']].copy()
df2_selected['郵遞區號'] = None
df2_selected = df2_selected.rename(columns={'Latitude': '緯度', 'Longitude': '經度'})

df3_selected = df3[['郵遞區號', '地區（所屬縣市名稱）', '熱點名稱', '地址', '緯度', '經度']].copy()
df3_selected.columns = ['郵遞區號', '地區', '熱點名稱', '地址', '緯度', '經度']

df1_merged = df1_selected.merge(df3_selected[['郵遞區號', '地址']], on='地址', how='left')
df2_merged = df2_selected.merge(df3_selected[['郵遞區號', '地址']], on='地址', how='left')

result = pd.concat([df1_merged, df2_merged], ignore_index=True)
result['郵遞區號'].fillna(result['郵遞區號_y'], inplace=True)
result['郵遞區號'] = result['郵遞區號'].combine_first(result['郵遞區號_x'])
result = result[['郵遞區號', '地區', '熱點名稱', '地址', '經度', '緯度']]

result = result.sort_values(by='郵遞區號').reset_index(drop=True)
```
