df1['資本總額'] = pd.to_numeric(df1['資本總額'].str.strip().replace('', pd.NA))

result['獎補助事項'] = result['獎補助事項'].str.replace('補助中小企業擴散行動支付場域及創新應用-', '', regex=False)

result = result.sort_values(by='資本總額', ascending=False).reset_index(drop=True)
