dfs = []
for df in [df1, df2]:
    left = df[['券商', '價格', '買進股數']].copy()
    right = df[['券商.1', '價格.1', '買進股數.1']].copy()
    right.columns = ['券商', '價格', '買進股數']
    all_df = pd.concat([left, right], ignore_index=True)
    all_df = all_df.dropna(subset=['券商', '價格', '買進股數'])
    all_df = all_df[all_df['買進股數'] > 0]
    all_df['券商'] = all_df['券商'].astype(str).str.replace('\u3000', '').str.strip()
    dfs.append(all_df)
all_data = pd.concat(dfs, ignore_index=True)
all_data['成交金額'] = all_data['價格'] * all_data['買進股數']
grouped = all_data.groupby('券商', as_index=False).agg(
    總買進股數=('買進股數', 'sum'),
    總買進成交金額=('成交金額', 'sum')
)
grouped['均價'] = grouped['總買進成交金額'] / grouped['總買進股數']
grouped['均價'] = grouped['均價'].round(1)
grouped['總買進股數'] = grouped['總買進股數'].round(1)
grouped['總買進成交金額'] = grouped['總買進成交金額'].round(1)
grouped['股票代號'] = '2454'
result = grouped[['券商', '均價', '總買進股數', '總買進成交金額', '股票代號']]
