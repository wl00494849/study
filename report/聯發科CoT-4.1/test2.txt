df1_main = df1[['券商', '價格', '買進股數']].copy()
df1_vice = df1[['券商.1', '價格.1', '買進股數.1']].copy()

df2_main = df2[['券商', '價格', '買進股數']].copy()

grouped['均價'] = (grouped['成交金額'] / grouped['買進股數']).round(1)
grouped['股票代號'] = '2454'
result = grouped[['券商', '均價', '買進股數', '成交金額', '股票代號']]
