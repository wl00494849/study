```python
df1_left = df1[['券商', '價格', '買進股數']].rename(columns={'券商':'券商', '價格':'價格', '買進股數':'買進股數'})
df1_right = df1[['券商.1', '價格.1', '買進股數.1']].rename(columns={'券商.1':'券商', '價格.1':'價格', '買進股數.1':'買進股數'})
df2_left = df2[['券商', '價格', '買進股數']].rename(columns={'券商':'券商', '價格':'價格', '買進股數':'買進股數'})
df2_right = df2[['券商.1', '價格.1', '買進股數.1']].rename(columns={'券商.1':'券商', '價格.1':'價格', '買進股數.1':'買進股數'})

all_data = (
    pd.concat([df1_left, df1_right, df2_left, df2_right], ignore_index=True)
    .dropna(subset=['券商', '價格', '買進股數'])
)
all_data = all_data[all_data['券商'].astype(str).str.strip() != '']
all_data = all_data[all_data['買進股數'] != 0]

all_data['成交金額'] = all_data['價格'] * all_data['買進股數']

grouped = (
    all_data.groupby('券商', as_index=False)
    .agg({'買進股數': 'sum', '成交金額': 'sum'})
)

grouped['均價'] = (grouped['成交金額'] / grouped['買進股數']).round(1)
grouped['股票代號'] = '2454'

result = grouped[['券商', '均價', '買進股數', '成交金額', '股票代號']]
result = result.rename(columns={'買進股數':'總買進股數','成交金額':'總買進成交金額'})
```
