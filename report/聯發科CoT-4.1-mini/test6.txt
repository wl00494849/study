df1_left = df1.loc[:, ['券商', '價格', '買進股數']].rename(columns={'券商':'券商','價格':'價格','買進股數':'買進股數'})
df1_right = df1.loc[:, ['券商.1', '價格.1', '買進股數.1']].rename(columns={'券商.1':'券商','價格.1':'價格','買進股數.1':'買進股數'})
df2_left = df2.loc[:, ['券商', '價格', '買進股數']].rename(columns={'券商':'券商','價格':'價格','買進股數':'買進股數'})
df2_right = df2.loc[:, ['券商.1', '價格.1', '買進股數.1']].rename(columns={'券商.1':'券商','價格.1':'價格','買進股數.1':'買進股數'})
concat_df = pd.concat([df1_left, df1_right, df2_left, df2_right], ignore_index=True)
concat_df = concat_df.dropna(subset=['券商', '價格', '買進股數'])
concat_df = concat_df[concat_df['買進股數'] > 0]
agg = concat_df.groupby('券商').apply(lambda x: pd.Series({
    '總買進股數': x['買進股數'].sum(),
    '總買進成交金額': (x['價格'] * x['買進股數']).sum()
})).reset_index()
agg['均價'] = (agg['總買進成交金額'] / agg['總買進股數']).round(1)
agg['總買進成交金額'] = agg['總買進成交金額'].round(1)
agg = agg.loc[agg['總買進股數'] > 0, ['券商', '均價', '總買進股數', '總買進成交金額']]
agg['股票代號'] = '2454'
result = agg[['券商', '均價', '總買進股數', '總買進成交金額', '股票代號']]
