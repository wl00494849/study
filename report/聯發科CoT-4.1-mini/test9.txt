```python
df1_partA = df1[['券商', '價格', '買進股數']].rename(columns={'券商':'券商','價格':'價格','買進股數':'買進股數'})
df1_partB = df1[['券商.1', '價格.1', '買進股數.1']].rename(columns={'券商.1':'券商','價格.1':'價格','買進股數.1':'買進股數'})
df2_partA = df2[['券商', '價格', '買進股數']].rename(columns={'券商':'券商','價格':'價格','買進股數':'買進股數'})
df2_partB = df2[['券商.1', '價格.1', '買進股數.1']].rename(columns={'券商.1':'券商','價格.1':'價格','買進股數.1':'買進股數'})
merged = pd.concat([df1_partA, df1_partB, df2_partA, df2_partB], ignore_index=True)
filtered = merged[merged['買進股數'] > 0]
grouped = filtered.groupby('券商').apply(lambda x: pd.Series({
    '均價': (x['價格'] * x['買進股數']).sum() / x['買進股數'].sum(),
    '總買進股數': x['買進股數'].sum()
})).reset_index()
grouped['總買進成交金額'] = grouped['均價'] * grouped['總買進股數']
grouped = grouped.round({'均價':1,'總買進股數':1,'總買進成交金額':1})
grouped['股票代號'] = '2454'
result = grouped[['券商','均價','總買進股數','總買進成交金額','股票代號']]
```
