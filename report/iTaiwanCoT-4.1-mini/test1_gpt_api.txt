df1_ = df1.rename(columns={'經度':'經度','緯度':'緯度','地區':'地區','熱點名稱':'熱點名稱','地址':'地址'})

df1_['地區'] = df1_['地區'].fillna('')

all_df = pd.concat([df1_[['郵遞區號','地區','熱點名稱','地址','經度','緯度']],
                    df2_[['郵遞區號','地區','熱點名稱','地址','經度','緯度']],
                    df3_[['郵遞區號','地區','熱點名稱','地址','經度','緯度']]],
                   ignore_index=True)

mask_missing_code = all_df['郵遞區號']==''
for idx in all_df[mask_missing_code].index:
    addr = all_df.at[idx,'地址']
    match = df3_[df3_['地址'].str.contains(addr[:10], na=False)]
    if not match.empty:
        all_df.at[idx,'郵遞區號'] = match.iloc[0]['郵遞區號']
        if pd.isna(all_df.at[idx,'地區']) or all_df.at[idx,'地區']=='':
            all_df.at[idx,'地區'] = match.iloc[0]['地區']

all_df['郵遞區號'] = all_df['郵遞區號'].replace('', pd.NA).dropna()
all_df['郵遞區號'] = all_df['郵遞區號'].astype(int)
result = all_df.sort_values('郵遞區號', na_position='last').reset_index(drop=True)
