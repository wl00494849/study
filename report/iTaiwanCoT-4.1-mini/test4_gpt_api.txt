import re
def extract_zip(address):
    if pd.isna(address):
        return pd.NA
    m = re.search(r'(\d{3,5})', address)
    return int(m.group(1)) if m else pd.NA

df1_proc = df1.copy()
df1_proc['郵遞區號'] = df1_proc['地址'].apply(extract_zip)
df1_proc = df1_proc.rename(columns={'地區':'地區','熱點名稱':'熱點名稱','地址':'地址','經度':'經度','緯度':'緯度'})
df1_proc = df1_proc[['郵遞區號','地區','熱點名稱','地址','經度','緯度']]

df2_proc = df2.copy()
df2_proc['郵遞區號'] = df2_proc['地址'].apply(extract_zip)
def extract_area(addr):
    if pd.isna(addr):
        return pd.NA
    match = re.search(r'(台[北中南東西]{0,2}市|新北市|桃園市|臺北市|高雄市|臺中市|彰化縣|南投縣|雲林縣|嘉義市|嘉義縣|屏東縣|宜蘭縣|花蓮縣|台東縣|澎湖縣|金門縣|連江縣)', addr)
    return match.group(1) if match else pd.NA
df2_proc['地區'] = df2_proc['地址'].apply(extract_area)
df2_proc = df2_proc.rename(columns={'地點':'熱點名稱','地址':'地址','Longitude':'經度','Latitude':'緯度'})
df2_proc = df2_proc[['郵遞區號','地區','熱點名稱','地址','經度','緯度']]

df3_proc = df3.copy()
df3_proc = df3_proc.rename(columns={'郵地區號':'郵遞區號','地區（所屬縣市名稱）':'地區','熱點名稱（依熱點所在地命名）':'熱點名稱','地址':'地址','經度':'經度','緯度':'緯度'})
df3_proc = df3_proc[['郵遞區號','地區','熱點名稱','地址','經度','緯度']]

result = pd.concat([df1_proc, df2_proc, df3_proc], ignore_index=True)
result['郵遞區號'] = pd.to_numeric(result['郵遞區號'], errors='coerce')
result = result.sort_values(by='郵遞區號', na_position='last').reset_index(drop=True)
