df1_ = df1.copy()

df2_renamed = df2_.rename(columns={'地點':'熱點名稱','地址':'地址','Longitude':'經度','Latitude':'緯度'})[['郵遞區號','地區','熱點名稱','地址','經度','緯度']]

df3_ = df3.copy()
df3_renamed = df3_.rename(columns={'郵地區號':'郵遞區號','地區（所屬縣市名稱）':'地區','熱點名稱（依熱點所在地命名）':'熱點名稱','地址':'地址','經度':'經度','緯度':'緯度'})[['郵遞區號','地區','熱點名稱','地址','經度','緯度']]

result = pd.concat([df1_renamed, df2_renamed, df3_renamed], ignore_index=True)
result['郵遞區號'] = result['郵遞區號'].astype('Int64')
result = result.sort_values(by='郵遞區號', na_position='last').reset_index(drop=True)
