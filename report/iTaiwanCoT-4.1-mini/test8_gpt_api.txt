```python
df2 = df2.rename(columns={'地點':'熱點名稱', 'Latitude':'緯度', 'Longitude':'經度'})
df3 = df3.rename(columns={
    '主管機關（所屬機關名稱）':'主管機關', 
    '地區（所屬縣市名稱）':'地區', 
    '熱點名稱（依熱點所在地命名）':'熱點名稱', 
    '郵地區號':'郵遞區號'
})
df1['郵遞區號'] = df1['地址'].str.extract(r'(\d{3})', expand=False)
df2['郵遞區號'] = df2['地址'].str.extract(r'(\d{3})', expand=False)
df3['郵遞區號'] = df3['郵遞區號'].astype(str).str.zfill(3)
df2['地區'] = df2['地址'].str.extract(r'(高雄市|雲林縣|南投縣)', expand=False)
columns = ['郵遞區號', '地區', '熱點名稱', '地址', '經度', '緯度']
df1_sel = df1.assign(經度=df1['經度'].astype(float), 緯度=df1['緯度'].astype(float))[columns]
df2_sel = df2.assign(經度=df2['經度'].astype(float), 緯度=df2['緯度'].astype(float))[columns]
df3_sel = df3.assign(經度=df3['經度'].astype(float), 緯度=df3['緯度'].astype(float))[columns]
result = pd.concat([df1_sel, df2_sel, df3_sel], ignore_index=True)
result['郵遞區號'] = pd.to_numeric(result['郵遞區號'], errors='coerce')
result = result.sort_values(by='郵遞區號', na_position='last').reset_index(drop=True)
```
